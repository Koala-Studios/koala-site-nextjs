async (page) => {
  const viewport = { width: 1440, height: 1100 };
  const replacements = [
    {
      url: "https://www.drinkfreezo.com/collections/make-at-home",
      output: "public/images/project/freezo/03-collection-make-at-home.jpg",
    },
    {
      url: "https://hopeharvest.ca/pages/about-us",
      output: "public/images/project/hope-harvest/06-about.jpg",
    },
  ];

  async function dismissAndClean() {
    await page.keyboard.press("Escape").catch(() => undefined);
    await page.waitForTimeout(300);
    await page.evaluate(() => {
      const closeText =
        /^(close|dismiss|no thanks|no, thanks|maybe later|not now|continue shopping|skip|x|×)$/i;
      const suspect =
        /(klaviyo|privy|justuno|smile|rewards|bucks|newsletter|popup|modal|gorgias|chat|attentive|postscript|spin|discount|coupon|notify|subscribe|sms|email)/i;

      function visible(element) {
        const style = getComputedStyle(element);
        const rect = element.getBoundingClientRect();
        return (
          style.display !== "none" &&
          style.visibility !== "hidden" &&
          Number(style.opacity) !== 0 &&
          rect.width > 4 &&
          rect.height > 4
        );
      }

      function label(element) {
        return [
          element.getAttribute("aria-label"),
          element.getAttribute("title"),
          element.textContent,
          typeof element.className === "string" ? element.className : "",
          element.id,
        ]
          .filter(Boolean)
          .join(" ");
      }

      for (const element of Array.from(document.querySelectorAll("button, [role='button'], a, [aria-label]")).reverse()) {
        if (!visible(element)) continue;
        const text = label(element).trim();
        if (closeText.test(text) || (/close|dismiss/i.test(text) && suspect.test(text))) {
          element.click();
        }
      }

      const viewportArea = window.innerWidth * window.innerHeight;
      for (const element of Array.from(document.querySelectorAll("body *")).reverse()) {
        if (!visible(element)) continue;
        const style = getComputedStyle(element);
        const rect = element.getBoundingClientRect();
        const width = Math.max(0, Math.min(rect.right, window.innerWidth) - Math.max(rect.left, 0));
        const height = Math.max(0, Math.min(rect.bottom, window.innerHeight) - Math.max(rect.top, 0));
        const areaRatio = (width * height) / viewportArea;
        const fixed = style.position === "fixed";
        const centered =
          rect.left < window.innerWidth * 0.75 &&
          rect.right > window.innerWidth * 0.25 &&
          rect.top < window.innerHeight * 0.75 &&
          rect.bottom > window.innerHeight * 0.18;
        const highZ = Number.parseInt(style.zIndex, 10) > 999;
        const text = label(element);
        const iframe = element.tagName === "IFRAME";

        if (
          (iframe && fixed && suspect.test(text)) ||
          (fixed && suspect.test(text)) ||
          (fixed && centered && highZ && areaRatio > 0.025) ||
          (fixed && centered && areaRatio > 0.12) ||
          (highZ && centered && areaRatio > 0.18)
        ) {
          element.remove();
        }
      }

      document.documentElement.style.overflow = "auto";
      document.body.style.overflow = "auto";
      if (document.activeElement instanceof HTMLElement) {
        document.activeElement.blur();
      }
    });
  }

  async function overlayReport() {
    return await page.evaluate(() => {
      const viewportArea = window.innerWidth * window.innerHeight;
      const suspect =
        /(congratulations|claim|spin|win|discount|coupon|bucks|rewards|notify me|newsletter|sign up|subscribe|sms|email)/i;
      return Array.from(document.querySelectorAll("body *"))
        .map((element) => {
          const style = getComputedStyle(element);
          const rect = element.getBoundingClientRect();
          if (
            style.display === "none" ||
            style.visibility === "hidden" ||
            Number(style.opacity) === 0 ||
            rect.width < 80 ||
            rect.height < 60
          ) {
            return null;
          }
          const width = Math.max(0, Math.min(rect.right, window.innerWidth) - Math.max(rect.left, 0));
          const height = Math.max(0, Math.min(rect.bottom, window.innerHeight) - Math.max(rect.top, 0));
          const areaRatio = (width * height) / viewportArea;
          const text = (element.innerText || element.getAttribute("aria-label") || "").slice(0, 140);
          const fixed = style.position === "fixed";
          const highZ = Number.parseInt(style.zIndex, 10) > 999;
          const centered =
            rect.left < window.innerWidth * 0.75 &&
            rect.right > window.innerWidth * 0.25 &&
            rect.top < window.innerHeight * 0.75 &&
            rect.bottom > window.innerHeight * 0.18;
          if ((fixed && suspect.test(text)) || (fixed && centered && areaRatio > 0.08) || (highZ && centered && areaRatio > 0.12)) {
            return { tag: element.tagName, areaRatio: Number(areaRatio.toFixed(3)), text };
          }
          return null;
        })
        .filter(Boolean)
        .slice(0, 10);
    });
  }

  const results = [];
  for (const item of replacements) {
    await page.context().clearCookies().catch(() => undefined);
    await page.setViewportSize(viewport);
    await page.goto(item.url, { waitUntil: "domcontentloaded", timeout: 45000 });
    await page.waitForLoadState("networkidle", { timeout: 9000 }).catch(() => undefined);
    await page.waitForTimeout(7000);
    await dismissAndClean();
    await page.waitForTimeout(900);
    await dismissAndClean();
    await page.evaluate(() => window.scrollTo(0, 0)).catch(() => undefined);
    await page.waitForTimeout(650);
    const overlaysBeforeShot = await overlayReport();
    await page.screenshot({
      path: item.output,
      type: "jpeg",
      quality: 82,
      fullPage: false,
      animations: "disabled",
      timeout: 45000,
    });
    results.push({
      url: item.url,
      output: item.output,
      overlaysBeforeShot,
      title: await page.title().catch(() => ""),
    });
  }

  return results;
}

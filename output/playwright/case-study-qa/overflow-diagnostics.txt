async (page) => {
  await page.setViewportSize({ width: 1440, height: 1100 });
  await page.goto("http://localhost:3037/", { waitUntil: "domcontentloaded" });
  await page.waitForTimeout(1200);
  await page.evaluate(() =>
    document
      .querySelector('section[aria-label="Brands we have built for"]')
      ?.scrollIntoView({ block: "center" })
  );
  await page.waitForTimeout(1200);
  return await page.evaluate(() => {
    const rightEdge = window.innerWidth;
    return Array.from(document.querySelectorAll("body *"))
      .map((el) => {
        const rect = el.getBoundingClientRect();
        return {
          className:
            typeof el.className === "string" ? el.className : String(el.className),
          tag: el.tagName,
          text: (el.textContent || "").trim().slice(0, 50),
          left: Math.round(rect.left),
          right: Math.round(rect.right),
          width: Math.round(rect.width),
          overflowRight: Math.round(rect.right - rightEdge),
        };
      })
      .filter((item) => item.overflowRight > 0 || item.left < 0)
      .sort((a, b) => b.overflowRight - a.overflowRight)
      .slice(0, 20);
  });
}
